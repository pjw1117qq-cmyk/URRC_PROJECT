cmake_minimum_required(VERSION 3.8)
project(team01_race)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
foreach(node perception decision control_adapter)
  add_executable(${node}_node src/${node}_node.cpp)
  target_compile_features(${node}_node PUBLIC cxx_std_17)
  target_include_directories(${node}_node PRIVATE include)
  ament_target_dependencies(${node}_node rclcpp sensor_msgs geometry_msgs std_msgs)
  install(TARGETS ${node}_node DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch config models scripts DESTINATION share/${PROJECT_NAME})
install(FILES race_entry.json DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_gtest(test_decision test/test_decision.cpp)
  target_compile_features(test_decision PUBLIC cxx_std_17)
  target_include_directories(test_decision PRIVATE include)
  ament_target_dependencies(test_decision sensor_msgs)
  ament_add_pytest_test(test_model test/test_model.py TIMEOUT 60)
  add_launch_test(test/test_race_start_integration.py TIMEOUT 30)
endif()
ament_package()
